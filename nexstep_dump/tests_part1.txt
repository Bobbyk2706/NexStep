"""Scripted stand-in for a real LLM: reads the passages it is given and
answers like a competent model would, citing real passage ids/quotes."""
from __future__ import annotations

import json
import re

from app.ai.providers.base import AIProvider


def passage_id(context: str, needle: str) -> str | None:
    for block in re.split(r"\n\n(?=\[P\d+ \|)", context):
        if needle in block:
            m = re.match(r"\[(P\d+) \|", block)
            if m:
                return m.group(1)
    return None


def fact(context: str, value, needle: str):
    return {"value": value, "quote": needle, "passage": passage_id(context, needle)}


EMPTY = {"value": None, "quote": None, "passage": None}


class ScriptedProvider(AIProvider):
    name = "scripted"

    def __init__(self, script=None):
        self.calls: list[str] = []
        self.script = script  # optional override: fn(task, context) -> dict|str

    def generate(self, *, messages, response_schema=None, model=None,
                 temperature=0.0, max_tokens=None):
        user = next(m["content"] for m in messages if m["role"] == "user")
        context = user.split("PASSAGES:\n", 1)[-1]
        task = ("identity" if "identify the examination" in user else
                "dates" if "extract the schedule" in user else "eligibility")
        self.calls.append(task)
        if self.script:
            out = self.script(task, context)
            return out if isinstance(out, str) else json.dumps(out)
        return json.dumps(getattr(self, f"_{task}")(context))

    def _identity(self, c):
        return {"exam_name": fact(c, "GATE 2027", "GATE 2027"),
                "conducting_body": fact(c, "IIT Madras", "IIT Madras"),
                "release_date": fact(c, "2026-08-27", "27th August 2026")}

    def _dates(self, c):
        return {
            "application_start_date": EMPTY,
            "application_end_date": fact(c, "2026-09-27", "Sunday, September 27, 2026"),
            "late_fee_end_date": fact(c, "2026-10-05", "Monday, October 5, 2026"),
            "exam_dates": [
                {"label": "Week 1", "start_date": "2027-02-06", "end_date": "2027-02-07",
                 "quote": "February 6 & 7, 2027", "passage": passage_id(c, "February 6 & 7, 2027")},
                {"label": "Week 2", "start_date": "2027-02-13", "end_date": "2027-02-14",
                 "quote": "February 13 & 14, 2027", "passage": passage_id(c, "February 13 & 14, 2027")},
                {"label": "Week 3", "start_date": "2027-02-20", "end_date": "2027-02-21",
                 "quote": "February 20 & 21, 2027", "passage": passage_id(c, "February 20 & 21, 2027")},
            ]}

    def _eligibility(self, c):
        q = "third or higher years of any undergraduate"
        return {
            "has_no_age_limit": True, "age_limits": [],
            "qualifications": [
                {"level": lvl, "field": "Engineering/Technology/Science/Commerce/Arts",
                 "quote": q, "passage": passage_id(c, q)}
                for lvl in ("Bachelor's", "Master's", "PhD")],
            "minimum_percentage": None, "minimum_cgpa": None,
            "score_quote": None, "score_passage": None,
            "nationalities": [], "nationality_quote": None, "nationality_passage": None,
            "work_experience": EMPTY,
            "other_requirements": [{"text": "Currently in the third year or higher of a UG program, or degree completed",
                                    "quote": q, "passage": passage_id(c, q)}]}

def _passage(number: int, text: str) -> Passage:
    return Passage(
        id=f"P{number}",
        index=number - 1,
        page_start=1,
        page_end=1,
        section="Eligibility",
        text=text,
        tokens=0,
    )


def _compile(draft, *passages):
    by_id = {p.id: p for p in passages}

    grounder = Grounder(
        passages=by_id,
        doc_dates=document_dates([p.text for p in passages]),
    )

    context = "\n\n".join(p.tagged() for p in passages)

    info, rules = compiler.compile_eligibility(draft, grounder, context)

    return info, rules, grounder


def _dob_rules(rules):
    return sorted(
        (r.operator, r.value)
        for g in rules.rule_groups
        for r in g.rules
        if r.attribute == "Date of Birth"
    )

def _upsc_draft(**overrides):
    quote = (
        "must have attained the age of 21 years and must not have attained "
        "the age of 32 years on the 1st of August, 2025"
    )

    values = dict(
        age_limits=[
            facts.AgeLimit(kind="minimum", years=21, as_on_date="2025-08-01",
                           quote=quote, passage="P1"),
            facts.AgeLimit(kind="maximum", years=32, as_on_date="2025-08-01",
                           quote=quote, passage="P1"),
            facts.AgeLimit(
                kind="maximum", years=37, as_on_date="2025-08-01",
                quote="up to a maximum of five years if a candidate belongs "
                      "to a Scheduled Caste or a Scheduled Tribe",
                passage="P2"),
            facts.AgeLimit(
                kind="maximum", years=35, as_on_date="2025-08-01",
                quote="in the case of candidates belonging to Other "
                      "Backward Classes",
                passage="P2"),
        ],
        qualifications=[facts.QualificationFact(level="Bachelor's")],
        nationalities=[
            "India", "Nepal", "Bhutan",
            "Indian origin (from Pakistan, Burma, Sri Lanka)",
        ],
        nationality_quote=(

            "of Nepal, or (c) a subject of Bhutan"
        ),
        nationality_passage="P3",
        work_experience=facts.Fact(),
    )

    values.update(overrides)

    return facts.EligibilityDraft(**values)

def test_upsc_age_limits_and_concessions():
    info, rules, grounder = _compile(
        _upsc_draft(),
        _passage(1, UPSC_AGE),
        _passage(2, UPSC_RELAXATION),
        _passage(3, UPSC_NATIONALITY),

    assert info.maximum_age == 31

    assert grounder.conflicts == []
    assert any("Age maximum 37" in text for text in info.other_requirements)
    assert any("Age maximum 35" in text for text in info.other_requirements)

def rules_of(r):
    def walk(g):
        for x in g.rules:
            yield (x.attribute, x.operator, x.value)
        for c in g.child_groups:
            yield from walk(c)
    return {x for g in r.extraction.eligibility_rules.rule_groups for x in walk(g)}
